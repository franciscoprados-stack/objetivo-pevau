                    className="w-6 h-6 rounded-full bg-yellow-200 border border-yellow-400 hover:scale-110 transition"
                    title="Resaltado Amarillo"
                  />
                  <button 
                    onClick={() => formatText('hiliteColor', '#bbf7d0')} 
                    className="w-6 h-6 rounded-full bg-emerald-200 border border-emerald-400 hover:scale-110 transition"
                    title="Resaltado Verde"
                  />
                  <button 
                    onClick={() => formatText('hiliteColor', '#fecdd3')} 
                    className="w-6 h-6 rounded-full bg-rose-200 border border-rose-400 hover:scale-110 transition"
                    title="Resaltado Rojo"
                  />
                </div>

                <div className="text-[11px] text-slate-400 italic font-serif">
                  * Edita el texto libremente. Tus anotaciones se conservan localmente.
                </div>
              </div>

              {/* ÁREA DE LECTURA Y EDICIÓN */}
              <div 
                ref={editorRef}
                className="flex-1 p-6 lg:p-10 overflow-y-auto prose prose-slate max-w-none focus:outline-none font-serif leading-relaxed text-slate-800 selection:bg-amber-200"
                contentEditable={true}
                suppressContentEditableWarning={true}
                dangerouslySetInnerHTML={{ __html: customTemarios[autorTemario] }}
                onBlur={(e) => setCustomTemarios(prev => ({ ...prev, [autorTemario]: e.currentTarget.innerHTML }))}
              />
            </div>

          </div>
        )}

        {/* ================= SECCIÓN 2: GLOSARIO NORMATIVO (100 CONCEPTOS) ================= */}
        {activeTab === 'glosario' && (
          <div className="flex-1 bg-white rounded-2xl shadow-sm border border-slate-200 flex flex-col overflow-hidden">
            
            {/* ENCABEZADO DE BÚSQUEDA Y SELECCIÓN DE AUTOR */}
            <div className="bg-slate-900 text-white p-4 flex flex-col md:flex-row justify-between items-center gap-4">
              <div className="flex flex-wrap gap-2 w-full md:w-auto">
                {[
                  { id: 'platon', label: 'Platón (25)' },
                  { id: 'descartes', label: 'Descartes (25)' },
                  { id: 'nietzsche', label: 'Nietzsche (25)' },
                  { id: 'ortega', label: 'Ortega y Gasset (25)' }
                ].map(autor => (
                  <button 
                    key={autor.id} 
                    onClick={() => setAutorGlosario(autor.id)} 
                    className={`px-4 py-2 rounded-xl font-bold text-xs uppercase transition-all ${
                      autorGlosario === autor.id 
                        ? 'bg-amber-500 text-slate-950 shadow-md' 
                        : 'bg-slate-800 text-slate-300 hover:bg-slate-700'
                    }`}
                  >
                    {autor.label}
                  </button>
                ))}
              </div>

              <div className="flex items-center gap-2 w-full md:w-auto">
                {/* BUSCADOR */}
                <div className="relative flex-1 md:w-64">
                  <Search size={16} className="absolute left-3 top-1/2 -translate-y-1/2 text-slate-400" />
                  <input 
                    type="text" 
                    placeholder="Filtrar concepto..."
                    value={searchTerm}
                    onChange={(e) => setSearchTerm(e.target.value)}
                    className="w-full bg-slate-800 text-white pl-9 pr-4 py-2 rounded-xl text-xs border border-slate-700 focus:outline-none focus:border-amber-500"
                  />
                </div>

                {/* BOTÓN AÑADIR TÉRMINO */}
                <button 
                  onClick={() => setShowAddTermModal(true)}
                  className="bg-amber-500 text-slate-950 p-2 rounded-xl font-bold hover:bg-amber-400 transition"
                  title="Añadir concepto al glosario"
                >
                  <Plus size={18} />
                </button>
              </div>
            </div>

            {/* LISTADO DE CONCEPTOS ESTILO DICCIONARIO ACADÉMICO */}
            <div className="flex-1 p-6 overflow-y-auto bg-slate-50">
              <div className="grid grid-cols-1 md:grid-cols-2 gap-4">
                {(glosarioState[autorGlosario] || [])
                  .filter(item => 
                    item.term.toLowerCase().includes(searchTerm.toLowerCase()) || 
                    item.def.toLowerCase().includes(searchTerm.toLowerCase())
                  )
                  .map((item, index) => (
                    <div 
                      key={index} 
                      className="bg-white p-5 rounded-2xl shadow-sm border border-slate-200 hover:border-amber-400 transition flex flex-col justify-between"
                    >
                      <div>
                        <div className="flex justify-between items-start mb-2">
                          <h3 className="text-base font-bold text-slate-900 font-serif">{item.term}</h3>
                          <span className="text-[10px] bg-amber-100 text-amber-800 font-bold px-2 py-0.5 rounded-md uppercase font-mono">Normativo</span>
                        </div>
                        <p className="text-slate-600 text-xs font-serif leading-relaxed">{item.def}</p>
                      </div>
                    </div>
                  ))}
              </div>
            </div>

          </div>
        )}

        {/* MODAL PARA AÑADIR CONCEPTO */}
        {showAddTermModal && (
          <div className="fixed inset-0 bg-slate-950/70 backdrop-blur-sm z-50 flex items-center justify-center p-4">
            <div className="bg-white rounded-2xl max-w-md w-full p-6 shadow-2xl border border-slate-200">
              <div className="flex justify-between items-center mb-4">
                <h3 className="font-bold text-slate-900 font-serif text-lg">Añadir Concepto ({autorGlosario.toUpperCase()})</h3>
                <button onClick={() => setShowAddTermModal(false)} className="text-slate-400 hover:text-slate-600">
                  <X size={20} />
                </button>
              </div>
              <form onSubmit={handleAddConcept} className="space-y-4">
                <div>
                  <label className="block text-xs font-bold text-slate-700 uppercase mb-1">Término Filosófico</label>
                  <input 
                    type="text" 
                    required 
                    value={newTerm} 
                    onChange={(e) => setNewTerm(e.target.value)}
                    placeholder="Ej. Res Cogitans"
                    className="w-full border rounded-xl p-3 text-sm focus:outline-none focus:border-amber-500"
                  />
                </div>
                <div>
                  <label className="block text-xs font-bold text-slate-700 uppercase mb-1">Definición Académica (3-4 líneas)</label>
                  <textarea 
                    required 
                    rows={4}
                    value={newDef} 
                    onChange={(e) => setNewDef(e.target.value)}
                    placeholder="Escribe la definición normativa precisa..."
                    className="w-full border rounded-xl p-3 text-sm focus:outline-none focus:border-amber-500 font-serif"
                  />
                </div>
                <div className="flex justify-end gap-2 pt-2">
                  <button 
                    type="button" 
                    onClick={() => setShowAddTermModal(false)} 
                    className="px-4 py-2 rounded-xl text-xs font-bold text-slate-600 hover:bg-slate-100"
                  >
                    Cancelar
                  </button>
                  <button 
                    type="submit" 
                    className="px-6 py-2 bg-slate-900 text-white rounded-xl text-xs font-bold hover:bg-slate-800 shadow"
                  >
                    Guardar Concepto
                  </button>
                </div>
              </form>
            </div>
          </div>
        )}

        {/* ================= SECCIÓN 3: ATLAS EXPERTO (28 ESQUEMAS) ================= */}
        {activeTab === 'atlas' && (
          <div className="flex-1 bg-white rounded-2xl shadow-sm border border-slate-200 flex flex-col overflow-hidden">
            
            {/* SELECCIÓN DE AUTOR Y NAVEGACIÓN DE ESQUEMAS */}
            <div className="bg-slate-900 text-white p-4 flex flex-col md:flex-row justify-between items-center gap-4">
              <div className="flex flex-wrap gap-2">
                {[
                  { id: 'platon', label: 'Platón (7)' },
                  { id: 'descartes', label: 'Descartes (7)' },
                  { id: 'nietzsche', label: 'Nietzsche (7)' },
                  { id: 'ortega', label: 'Ortega y Gasset (7)' }
                ].map(autor => (
                  <button 
                    key={autor.id} 
                    onClick={() => { setAutorAtlas(autor.id); setEsquemaIndex(0); }} 
                    className={`px-4 py-2 rounded-xl font-bold text-xs uppercase transition-all ${
                      autorAtlas === autor.id 
                        ? 'bg-amber-500 text-slate-950 shadow-md' 
                        : 'bg-slate-800 text-slate-300 hover:bg-slate-700'
                    }`}
                  >
                    {autor.label}
                  </button>
                ))}
              </div>

              {/* CONTROLES ANTERIOR / SIGUIENTE */}
              <div className="flex items-center gap-3">
                <span className="text-xs font-mono text-slate-400">
                  Esquema {esquemaIndex + 1} de {ATLAS_COMPLETO[autorAtlas]?.length || 7}
                </span>
                <button 
                  onClick={() => setEsquemaIndex((esquemaIndex - 1 + ATLAS_COMPLETO[autorAtlas].length) % ATLAS_COMPLETO[autorAtlas].length)}
                  className="p-2 bg-slate-800 hover:bg-slate-700 rounded-xl text-white transition"
                >
                  <ChevronLeft size={18} />
                </button>
                <button 
                  onClick={() => setEsquemaIndex((esquemaIndex + 1) % ATLAS_COMPLETO[autorAtlas].length)}
                  className="p-2 bg-slate-800 hover:bg-slate-700 rounded-xl text-white transition"
                >
                  <ChevronRight size={18} />
                </button>
              </div>
            </div>

            {/* ÁREA DE VISUALIZACIÓN DEL ESQUEMA CSS */}
            <div className="flex-1 p-6 bg-slate-100 flex items-center justify-center overflow-y-auto">
              {(() => {
                const esquema = ATLAS_COMPLETO[autorAtlas]?.[esquemaIndex];
                if (!esquema) return null;

                return (
                  <div className="bg-white p-8 rounded-2xl shadow-xl border border-slate-200 w-full max-w-4xl">
                    <h3 className="text-xl font-bold font-serif text-slate-900 mb-6 border-b pb-3 text-center border-amber-200">
                      {esquema.title}
                    </h3>

                    {/* TIPO 1: TABLA COMPLEJA */}
                    {esquema.type === 'tabla_compleja' && (
                      <div className="overflow-x-auto">
                        <table className="w-full text-left border-collapse">
                          <thead>
                            <tr className="bg-slate-900 text-white">
                              {esquema.headers.map((h, i) => (
                                <th key={i} className="p-4 font-bold text-xs uppercase tracking-wider border border-slate-800">{h}</th>
                              ))}
                            </tr>
                          </thead>
                          <tbody>
                            {esquema.rows.map((row, i) => (
                              <tr key={i} className={i % 2 === 0 ? 'bg-slate-50' : 'bg-white'}>
                                {row.map((cell, j) => (
                                  <td key={j} className="p-4 border border-slate-200 font-serif text-xs text-slate-800 leading-relaxed">{cell}</td>
                                ))}
                              </tr>
                            ))}
                          </tbody>
                        </table>
                      </div>
                    )}

                    {/* TIPO 2: TRIPARTITO */}
                    {esquema.type === 'tripartito' && (
                      <div className="grid grid-cols-1 md:grid-cols-3 gap-6">
                        {[esquema.col1, esquema.col2, esquema.col3].map((col, idx) => (
                          <div key={idx} className="bg-slate-50 p-6 rounded-2xl border border-slate-200 flex flex-col justify-between">
                            <div>
                              <h4 className="font-bold text-amber-700 text-sm mb-4 border-b pb-2 font-serif">{col.title}</h4>
                              <ul className="space-y-2">
                                {col.items.map((item, i) => (
                                  <li key={i} className="text-xs font-serif text-slate-700 flex items-start gap-2">
                                    <span className="text-amber-500 font-bold">•</span>
                                    <span>{item}</span>
                                  </li>
                                ))}
                              </ul>
                            </div>
                          </div>
                        ))}
                      </div>
                    )}

                    {/* TIPO 3: PROCESO VERTICAL */}
                    {esquema.type === 'proceso_vertical' && (
                      <div className="flex flex-col gap-3">
                        {esquema.steps.map((step, idx) => (
                          <div key={idx} className="flex items-center gap-4 bg-slate-50 p-4 rounded-xl border border-slate-200">
                            <div className="w-8 h-8 rounded-full bg-slate-900 text-amber-400 font-bold flex items-center justify-center text-xs shrink-0">
                              {idx + 1}
                            </div>
                            <p className="font-serif text-xs text-slate-800 leading-relaxed">{step}</p>
                          </div>
                        ))}
                      </div>
                    )}

                  </div>
                );
              })()}
            </div>

          </div>
        )}

        {/* ================= SECCIÓN 4: SIMULADOR PEVAU (EXAMEN OFICIAL) ================= */}
        {activeTab === 'simulador' && (
          <div className="h-[80vh] flex flex-col">
            
            {/* ETAPA 1: SELECCIÓN DE EXAMEN */}
            {simuladorStep === 'seleccion' && (
              <div className="flex-1 flex flex-col items-center justify-center bg-white rounded-2xl shadow-sm border border-slate-200 p-8 text-center">
                <div className="w-20 h-20 bg-amber-100 text-amber-600 rounded-full flex items-center justify-center mb-6">
                  <ShieldCheck size={48} />
                </div>
                <h2 className="text-3xl font-serif font-bold text-slate-900 mb-2">Simulador Oficial PEvAU</h2>
                <p className="text-slate-600 max-w-md mb-8 text-sm leading-relaxed">
                  Entrena con pruebas aleatorias estructuradas exactamente según el modelo oficial de examen (Parte I: Disertación + Parte II: Cuestiones).
                </p>

                <div className="flex flex-col sm:flex-row gap-4">
                  <button 
                    onClick={() => iniciarExamen()}
                    className="px-8 py-4 bg-slate-900 hover:bg-slate-800 text-white font-bold rounded-2xl shadow-lg transition flex items-center justify-center gap-3"
                  >
                    <Dices size={20} />
                    <span>Generar Examen Aleatorio</span>
                  </button>
                </div>
              </div>
            )}

            {/* ETAPA 2: RESOLUCIÓN DEL EXAMEN */}
            {simuladorStep === 'resolviendo' && examenSeleccionado && (
              <div className="flex-1 flex flex-col lg:flex-row gap-6 overflow-hidden">
                
                {/* ENUNCIADO DEL EXAMEN (IZQUIERDA) */}
                <div className="lg:w-1/2 bg-white rounded-2xl shadow-sm border border-slate-200 p-6 overflow-y-auto">
                  <div className="bg-amber-50 border-l-4 border-amber-500 p-4 rounded-r-xl mb-6">
                    <span className="text-[10px] font-bold uppercase tracking-wider text-amber-800 bg-amber-200 px-2 py-0.5 rounded font-mono">PARTE I • DISERTACIÓN (2,5 PUNTOS)</span>
                    <p className="font-serif text-sm font-bold text-slate-900 mt-2 leading-relaxed">
                      {examenSeleccionado.disertacion}
                    </p>
                  </div>

                  <div className="bg-slate-50 p-6 rounded-2xl border border-slate-200 mb-6">
                    <span className="text-[10px] font-bold uppercase tracking-wider text-slate-600 bg-slate-200 px-2 py-0.5 rounded font-mono mb-3 inline-block">PARTE II • TEXTO Y CUESTIONES (7,5 PUNTOS)</span>
                    <blockquote className="border-l-4 border-slate-400 pl-4 italic text-slate-700 font-serif leading-relaxed text-sm mb-3">
                      "{examenSeleccionado.texto}"
                    </blockquote>
                    <p className="text-right text-xs font-bold text-slate-400 font-serif">{examenSeleccionado.obraCita}</p>
                  </div>

                  <div className="space-y-3 bg-white p-4 rounded-xl border border-slate-200">
                    <h4 className="text-xs font-bold text-slate-900 uppercase">Cuestiones Normativas:</h4>
                    <p className="text-xs font-serif text-slate-700">{examenSeleccionado.cuestion1}</p>
                    <p className="text-xs font-serif text-slate-700">{examenSeleccionado.cuestion2}</p>
                    <p className="text-xs font-serif text-slate-700">{examenSeleccionado.cuestion3}</p>
                  </div>
                </div>

                {/* CAJA DE RESPUESTA DEL ALUMNO (DERECHA) */}
                <div className="lg:w-1/2 bg-white rounded-2xl shadow-sm border border-slate-200 flex flex-col overflow-hidden">
                  <div className="bg-slate-900 text-white px-6 py-3 flex justify-between items-center">
                    <span className="font-bold text-xs uppercase tracking-wider">Tu Ejercicio Escrito</span>
                    <div className="flex items-center gap-4">
                      <label className="flex items-center gap-1.5 text-[11px] text-slate-300 cursor-pointer">
                        <input 
                          type="checkbox" 
                          checked={penalizacionOrtografia} 
                          onChange={(e) => setPenalizacionOrtografia(e.target.checked)} 
                          className="rounded text-amber-500 focus:ring-0"
                        />
                        <span>Simular Penalización Ortográfica (-1.0)</span>
                      </label>
                      <button 
                        onClick={entregarExamen}
                        className="px-4 py-1.5 bg-amber-500 hover:bg-amber-400 text-slate-950 font-bold text-xs rounded-xl shadow transition"
                      >
                        Entregar Examen
                      </button>
                    </div>
                  </div>

                  <textarea 
                    value={respuestaAlumno}
                    onChange={(e) => setRespuestaAlumno(e.target.value)}
                    className="flex-1 p-6 outline-none resize-none font-serif text-slate-800 text-sm leading-relaxed bg-slate-50 focus:bg-white transition"
                    placeholder="Escribe aquí tu disertación completa y las tres cuestiones argumentadas..."
                  />
                  
                  <div className="bg-slate-100 px-6 py-2 border-t text-right text-xs font-mono text-slate-500">
                    Palabras redactadas: {respuestaAlumno.trim().split(/\s+/).filter(w => w.length > 0).length}
                  </div>
                </div>

              </div>
            )}

            {/* ETAPA 3: DICTAMEN DE REVISIÓN E IA */}
            {simuladorStep === 'revision' && (
              <div className="flex-1 bg-white rounded-2xl shadow-sm border border-slate-200 p-8 flex flex-col items-center overflow-y-auto">
                
                <div className="w-16 h-16 bg-amber-100 text-amber-600 rounded-full flex items-center justify-center mb-4">
                  <Sparkles size={32} />
                </div>
                
                <h2 className="text-2xl font-serif font-bold text-slate-900 mb-1">Dictamen de Evaluación PEvAU</h2>
                <p className="text-xs text-slate-500 mb-6">Calificación Algorítmica + Modelo Transversal de Excelencia</p>

                {/* NOTA NUMÉRICA FINAL */}
                <div className="bg-slate-900 text-white rounded-2xl px-10 py-6 text-center mb-8 shadow-xl">
                  <span className="text-xs uppercase font-mono text-slate-400 block mb-1">Calificación Estimada</span>
                  <span className="text-5xl font-extrabold text-amber-400 font-serif">{notaFinal.toFixed(2)}</span>
                  <span className="text-sm font-bold text-slate-400"> / 10.00</span>
                  {notaFinal === 0.0 && (
                    <p className="text-rose-400 text-xs mt-2 font-serif">
                      * Calificación 0.0 asignada obligatoriamente por respuesta vacía o inferior a 20 palabras.
                    </p>
                  )}
                </div>

                {/* MODELO IDEAL 10/10 */}
                <div className="w-full max-w-4xl bg-slate-50 border border-slate-200 rounded-2xl p-6 text-left">
                  <h3 className="font-bold text-slate-900 font-serif text-base mb-4 border-b pb-2 flex items-center gap-2">
                    <CheckCircle size={18} className="text-emerald-600" />
                    <span>Modelo de Respuesta 10/10 y Análisis Transversal</span>
                  </h3>
                  <div className="whitespace-pre-line font-serif text-slate-700 text-sm leading-relaxed bg-white p-6 rounded-xl border border-slate-200">
                    {examenSeleccionado?.idealIA}
                  </div>
                </div>

                <button 
                  onClick={() => setSimuladorStep('seleccion')}
                  className="mt-8 px-8 py-3 bg-slate-900 hover:bg-slate-800 text-white font-bold rounded-2xl shadow transition"
                >
                  Realizar Otro Simulacro
                </button>

              </div>
            )}

          </div>
        )}

      </main>

      {/* FOOTER OFICIAL DE AUTORÍA */}
      <footer className="bg-slate-950 text-slate-500 text-xs py-6 border-t border-slate-900 text-center">
        <div className="max-w-7xl mx-auto px-4 flex flex-col sm:flex-row justify-between items-center gap-2">
          <p className="font-bold text-slate-300">© Francisco Prados | Plataforma PEvAU</p>
          <p className="text-[11px]">Historia de la Filosofía • Bachillerato & Preparación Universitaria</p>
        </div>
      </footer>

    </div>
  );
}
